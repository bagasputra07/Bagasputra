🎯 Pengantar Generic

Apa itu Generic? Fitur untuk membuat kode yang bekerja dengan berbagai tipe data.

Analog: Kotak penyimpanan yang bisa menampung apa saja, tapi dengan label tipe.

📌 Manfaat Generic:

✅ Type safety (keamanan tipe)   ✅ Code reusability (penggunaan ulang)   ✅ Compile-time checking

⚠️ Masalah Tanpa Generic

class Box<T> {
  T data;
  Box(this.data);

void main() {
  var intBox = Box<int>(42);           // T = int
  var stringBox = Box<String>('Hello'); // T = String

  int value = intBox.data;              // ✅ Aman, pasti int
  // stringBox.data = 123;              // ❌ ERROR compile-time (tidak bisa)

  print(intBox.data);   // Output: 42
  print(stringBox.data); // Output: Hello
  print(value);          // Output: 42
}
✅ OUTPUT

// 1. Class Repository<T>
class Repository<T> {
  final Map<int, T> _storage = {};
  int _nextId = 1;

  int save(T item) {
    int id = _nextId++;
    _storage[id] = item;
    print('✅ Saved: ID $id → $item');
    return id;
  }

  T? getById(int id) {
    return _storage[id];
  }

  List<T> getAll() {
    return _storage.values.toList();
  }

  void printAll() {
    if (_storage.isEmpty) {
      print('📭 Repository kosong');
      return;
    }
    print('📦 Isi Repository:');
    _storage.forEach((id, item) => print('   ID $id: $item'));
  }
}

// 2. Fungsi swap
void swap<T>(List<T> list, int i, int j) {
  if (i < 0 || i >= list.length || j < 0 || j >= list.length) {
    print('❌ Error: Indeks tidak valid!');
    return;
  }
  if (i == j) return;

  T temp = list[i];
  list[i] = list[j];
  list[j] = temp;
}

// 3. Class Triple
class Triple<A, B, C> {
  A first;
  B second;
  C third;

  Triple(this.first, this.second, this.third);

  A getFirst() => first;
  B getSecond() => second;
  C getThird() => third;

  void setFirst(A value) => first = value;
  void setSecond(B value) => second = value;
  void setThird(C value) => third = value;

  void printTriple() {
    print('($first, $second, $third)');
  }

  List toList() => [first, second, third];
}

class Product {
  final int id; final String name; final double price;
  Product(this.id, this.name, this.price);
  @override String toString() => 'Product($id, $name, $price)';
}

class User {
  final int id; final String username;
  User(this.id, this.username);
  @override String toString() => 'User($id, $username)';
}

class GenericRepository {
  final Map _items = {};
  int _nextId = 1;

  T? create(T item) { _items[_nextId] = item; return item; }
  T? read(int id) => _items[id];
  T? update(int id, T item) { if (_items.containsKey(id)) { _items[id] = item; return item; } return null; }
  bool delete(int id) => _items.remove(id) != null;
  List readAll() => _items.values.toList();
}

void main() {
  var productRepo = GenericRepository();
  productRepo.create(Product(1, 'Laptop', 15000000));
  productRepo.create(Product(2, 'Mouse', 250000));
  print('Products: ${productRepo.readAll()}');

  var userRepo = GenericRepository();
  userRepo.create(User(1, 'PRABOWO'));
  userRepo.create(User(2, 'GIBRAN'));
  print('Users: ${userRepo.readAll()}');
}

📖 Ringkasan

🎯 Poin Penting:

  1. Generic membuat kode type-safe dan reusable
  2. Bisa digunakan di class, fungsi, collections
  3. Type inference memudahkan penggunaan
  4. Constraints (extends) membatasi tipe yang diperbolehkan

📌 Best Practice:

  • Gunakan generic untuk menghindari dynamic
  • Gunakan extends untuk membatasi tipe
  • Manfaatkan type inference untuk kode lebih bersih
